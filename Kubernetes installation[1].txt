---First install python pip version
sudo yum install -y python3-pip

---install AWS CLI
curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
unzip awscliv2.zip
sudo ./aws/install

--------------------Installl kops

curl -Lo kops https://github.com/kubernetes/kops/releases/download/$(curl -s https://api.github.com/repos/kubernetes/kops/releases/latest | grep tag_name | cut -d '"' -f 4)/kops-linux-amd64
chmod +x kops
sudo mv kops /usr/local/bin/kops

-------------------Do AWS configure
Go to IAM<-->My security connections<---> create key

Access key
Secret access key
Default region
Default ouput format --Json

------------------Create a bucket
create s3 bucket 
aws s3api create-bucket \
  --bucket vamsi-kube-123 \
  --region us-east-2 \
  --create-bucket-configuration LocationConstraint=us-east-2

---------------------Create cluster

kops create cluster \
  --name=testk8scluster.k8s.local \
  --state=s3://vamsi-kube-123 \
  --zones=us-east-2a \
  --node-count=1 \
  --node-size=t2.medium \
  --master-size=t2.medium \
  --master-volume-size=20 \
  --node-volume-size=20
    
-------Do Export

export KOPS_STATE_STORE=s3://vamsi-kube

------Do update cluster
 kops update cluster --name=vamsik8scluster.k8s.local --yes --admin
 
 kops update cluster --name=testk8scluster.k8s.local --state=s3://vamsi-kube-123 --yes

 
 
 ----Do validate 
 kops validate cluster --name=vamsik8scluster.k8s.local

 kops delete cluster --name vamsik8scluster.k8s.local --state s3://vamsi-kube-123 --yes
